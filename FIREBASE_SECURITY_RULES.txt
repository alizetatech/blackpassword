// BLACKPASSWORD - REGLAS DE SEGURIDAD v12.0-ZERO-TRUST
// Copia y pega esto en Firestore -> Rules.
// Estas reglas validan la identidad cruzada y el estado de la cuenta mediante Custom Claims.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Función para validar que el token tiene los claims de Blackpassword y la cuenta está activa
    function isVerifiedIdentity(emailHash) {
      return request.auth != null && 
             request.auth.token.email_hash == emailHash &&
             request.auth.token.role == 'USER' &&
             request.auth.token.account_status == 'ACTIVE';
    }

    // Colección de Usuarios (Document ID = Email Hash)
    match /usuarios/{emailHash} {
      allow read, write: if isVerifiedIdentity(emailHash);

      // Sub-colecciones blindadas
      match /boveda/{secretId} {
        allow read, write: if isVerifiedIdentity(emailHash);
      }

      match /archivos/{fileId} {
        allow read, write: if isVerifiedIdentity(emailHash);
      }
      
      match /imagenes/{imgId} {
        allow read, write: if isVerifiedIdentity(emailHash);
      }
    }

    // Bitácora de Auditoría: Solo el dueño o un ADMIN puede leer
    match /audit_logs/{logId} {
      allow create: if request.auth != null && request.auth.token.account_status == 'ACTIVE';
      allow read: if request.auth != null && 
                   (request.auth.token.email_hash == resource.data.email_hash || 
                    request.auth.token.role == 'ADMIN');
      allow update, delete: if false;
    }

    // Alertas de Intrusión: Solo ADMIN puede gestionar
    match /intrusion_alerts/{alertId} {
      allow create: if true; 
      allow read, update: if request.auth != null && request.auth.token.role == 'ADMIN';
      allow delete: if false;
    }

    // Bloqueo Global por Defecto
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
