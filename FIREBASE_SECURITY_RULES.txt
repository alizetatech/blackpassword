rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- FUNCIONES DE AYUDA ---
    function isAuthenticated() {
      return request.auth != null;
    }

    // Verifica que el hash del documento coincida con el hash en el token del usuario
    function isOwner(userHash) {
      return isAuthenticated() && request.auth.token.email_hash == userHash;
    }

    // --- COLECCIÓN DE USUARIOS (Hashed: "usuarios") ---
    match /1f4ade47160c389903c1a9e19d2a6739681fa8955f1c7353084e638675297b5d/{userHash} {
      // Solo el dueño del hash puede leer/escribir su perfil
      allow read, write: if isOwner(userHash);

      // Sub-colección BÓVEDA (Hashed: "boveda")
      match /77bcededfba243bed38327a48293c4b61267ccc9ef5a7c374f709a81fcf60031/{secretId} {
        allow read, write: if isOwner(userHash);
      }
      
      // Sub-colección ARCHIVOS (Hashed: "archivos")
      match /3340292fcf65c3a26b646a82989d04d8b680bee67b9aab516ac0622cdc3e80e0/{fileId} {
        allow read, write: if isOwner(userHash);
      }
      
      // Sub-colección IMÁGENES (Literal: "imagenes")
      match /imagenes/{imgId} {
        allow read, write: if isOwner(userHash);
      }
    }

    // --- LOGS DE AUDITORÍA ---
    match /audit_logs/{logId} {
      allow create: if isAuthenticated();
      allow read: if false; // Solo administradores desde la consola
    }

    // --- ALERTAS DE INTRUSIÓN ---
    match /intrusion_alerts/{alertId} {
      allow create: if true; // Cualquier intento fallido debe poder registrarse
      allow read, write: if false;
    }
  }
}
