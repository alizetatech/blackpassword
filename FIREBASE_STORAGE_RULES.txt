// Copia y pega esto en Storage -> Rules.

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Función de validación de ruta basada en el email_hash del token
    function ownsFolder(emailHash) {
      return request.auth != null && request.auth.token.email_hash == emailHash;
    }

    // Imágenes: Solo el dueño del hash puede acceder
    match /imagenes/{emailHash}/{imageId} {
      allow read, write: if ownsFolder(emailHash);
    }

    // Archivos: Solo el dueño del hash puede acceder
    match /archivos/{emailHash}/{fileId} {
      allow read, write: if ownsFolder(emailHash);
    }

    // Bloqueo total de lo demás
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
