// Copia y pega esto en Storage -> Rules.

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Función de verificación de identidad y estado
    function isVerified(emailHash) {
      return request.auth != null && 
             request.auth.token.email_hash == emailHash &&
             request.auth.token.account_status == 'ACTIVE';
    }

    // Carpeta de Imágenes: Solo el dueño lee y escribe
    match /imagenes/{emailHash}/{imageId} {
      allow read, write: if isVerified(emailHash);
    }

    // Carpeta de Archivos: Solo el dueño lee y escribe
    match /archivos/{emailHash}/{fileId} {
      allow read, write: if isVerified(emailHash);
    }

    // Carpeta de Perfiles: Solo el dueño gestiona su avatar
    match /perfiles/{emailHash}/{photoId} {
      allow read: if request.auth != null; // Cualquiera logueado puede ver el avatar
      allow write: if isVerified(emailHash);
    }

    // Bloqueo global de lo demás
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
